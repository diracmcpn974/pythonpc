    Thème 1 : Electromagnétisme

    Champ électrostatique \(\overrightarrow{E}\) créé par un ensemble de charges ponctuelles

    Soit une distribution de \(N\) charges ponctuelles \(q_i\) positionnées aux points \(P_i\), le champ électrostatique \(\overrightarrow{E}\) crée en un point \(M\) s'écrit :
    \(\overrightarrow{E}(M)= \displaystyle\sum_{i=1}^{N} \overrightarrow{E_i}(M) = \displaystyle\sum_{i=1}^{N} \frac{q_i}{4\pi\epsilon_0} \frac{\overrightarrow{P_iM}}{P_iM^3}\)

    Les lignes de champ sont les courbes telles qu'en tout point la direction du champ y soit tangente.
    Les surfaces équipotentielles sont l'ensemble des points où le potentiel est constant. Elles sont donc perpendiculaires aux lignes de champ.

    import numpy as np
import matplotlib.pyplot as plt

#Liste de charges Qi et positionnée au point Pi de coordonnées xi,yi 
charge=[(1.602e-19,-1,0),(1.602e-19,1,0),(1.602e-19,0,1),(1.602e-19,0,-1)]

#Grille carré de points Mi où le champ total E sera tracé
N=50 #Nombre de points sur une ligne ou sur une colonne
xM=np.linspace(-2,2,N)
yM=np.linspace(-2,2,N)
XM,YM=np.meshgrid(xM,yM) #XM et YM contient les abscisses et les ordonnées des points Mi de la grille

e0=8.854e-12 #Permittivité du vide en F/m
EX=np.zeros((N,N))#Tableau NxN des abscisses du champ total E
EY=np.zeros((N,N))#Tableau NxN des ordonnées du champ total E

#Pour chaque charge Qi en Pi
for q,x,y in charge:
    #Calcul des coordonnées des vecteurs PiMi 
    X=XM-x #abscisses des vecteurs PiMi
    Y=YM-y #ordonnées des vecteurs PiMi
    #Calcul de la norme des vecteurs PiMi
    R=np.sqrt(X**2+Y**2)
    #On évite d'être proche de qi
    R[R<0.7]=np.nan
    #Calcul des composantes du champ Ei de la charge Qi
    Ex = q/(4*np.pi*e0*R**3)*X #Abscisse de Ei
    Ey = q/(4*np.pi*e0*R**3)*Y #Ordonnée de Ei
    EX = EX+Ex
    EY = EY+Ey

fig,ax=plt.subplots()

ax.set_aspect("equal") #repère orthonormé
xP=[pos[1]for pos in charge] #Abscisse de Pi
yP=[pos[2]for pos in charge] #Ordonnée de Pi
ax.scatter(xP,yP) #Tracé des charges ponctuelles
c = ax.quiver(XM,YM,EX,EY,color="red", angles='xy', scale=6e-9, scale_units='xy', units='dots') #Tracé des vecteurs champ electrique

plt.show()

    import numpy as np
import matplotlib.pyplot as plt

#Liste de charges Qi positionnées au point Pi de coordonnées xi,yi 
charge=[(1.602e-19,-1,0),(1.602e-19,1,0),(1.602e-19,0,1),(1.602e-19,0,-1)]

#Grille carré de points Mi où le potentiel V sera calculé
N=100 #Nombre de points sur une ligne ou sur une colonne
xM=np.linspace(-2,2,N)
yM=np.linspace(-2,2,N)
XM,YM=np.meshgrid(xM,yM) #XM et YM contient les abscisses et les ordonnées des points Mi de la grille

e0=8.854e-12 #Permittivité du vide en F/m
V=np.zeros((N,N))#Tableau NxN des potentiels V

#Pour chaque charge Qi en Pi
for q,x,y in charge:
    #Calcul des coordonnées des vecteurs PiMi 
    X=XM-x #abscisses des vecteurs PiMi
    Y=YM-y #ordonnées des vecteurs PiMi
    #Calcul de la norme des vecteurs PiMi
    R=np.sqrt(X**2+Y**2)
    #Calcul des potentiels Vi de la charge Qi
    Vi = q/(4*np.pi*e0*R)
    V=V+Vi

fig,ax=plt.subplots()

ax.set_aspect("equal") #repère orthonormé
xP=[pos[1]for pos in charge] #Abscisse de Pi
yP=[pos[2]for pos in charge] #Ordonnée de Pi
ax.scatter(xP,yP) #Tracé des charges ponctuelles

c=ax.contour(XM, YM, V, levels=500) # Tracé des courbes équipotentielles
fig.colorbar(c, ax=ax,label="Potentiel (V)")

plt.show()

    Thème 2 : Thermodynamique

    Equation de Van Der Waals : Isothermes du méthane

    L'équation d'état de Van Der Waals pour un gaz réel s'écrit sous sa forme intensive : \(P = \frac{RT}{v-b}-\frac{a^{2}}{v^{2}}\)

    avec \(P\) pression, \(R\) constante des gaz parfaits, \(T\) température, \(a\) terme de cohésion, \(b\) covolume molaire et \(v\) volume molaire

    import numpy as np
import matplotlib.pyplot as plt
import matplotlib.colors as mcolors

#Paramètres pour le méthane
a = 2.253 #L^2.atm.mol^-2
b = 0.04278 #L.mol^-1
R = 0.08206 #L.atm.K^-1.mol^-1

temperature=np.linspace(162,210,40) # Température entre 162K et 210K
v=np.linspace(1.5*b,9*b,100) #Volume molaire entre 1,5b et 9b

fig,ax = plt.subplots()

ax.set_xlabel(r"$\text{v volume molaire } (L.mol^{-1})$")
ax.set_ylabel(r"$\text{P pression } (atm)$")
ax.set_title("Courbes isothermes du méthane")

cmap = plt.colormaps["coolwarm_r"]
norm = mcolors.Normalize(vmin=temperature.min(), vmax=temperature.max())

for T in temperature:
P=R*T/(v-b)-a/(v**2)
ax.plot(v,P,color=cmap(norm(T)))

cbar = fig.colorbar(plt.cm.ScalarMappable(norm=norm, cmap=cmap), ax=ax)
cbar.set_label("Température (K)")
plt.show()

    Interférences : ondes mécaniques

    On lache 2 objets dans l'eau. Cela produit une onde dont l'amplitude peut être modélisée par l'équation suivante :
    \(S(\overrightarrow{r}) = A sin (k|\overrightarrow{r} − \overrightarrow{r_0}|) + A sin (k|\overrightarrow{r} − \overrightarrow{r_1}|)\)

    avec \(A\) amplitude, \(k\) vecteur d'onde, \(\overrightarrow{r}\) vecteur position où l'onde est observé, \(\overrightarrow{r_0}\) et \(\overrightarrow{r_1}\) vecteurs position des 2 objets.

import matplotlib.pyplot as plt

#2 Objets lachés en r0(x0,y0) et r1(x1,y1)
x0,x1=-1,1
y0,y1=0,0 

#Grille carré des positions ri
N=500 #Nombre de points sur une ligne ou sur une colonne
xi=np.linspace(-4,4,N)
yi=np.linspace(-4,4,N)
Xi,Yi=np.meshgrid(xi,yi) 

A=1
k=2*np.pi/0.3
V=np.zeros((N,N))#Tableau NxN des abscisses du champ total E

delta_r0=np.sqrt((Xi-x0)**2+(Yi-y0)**2) #Norme de r-r0

W=A*np.sin(k*delta_r0)+A*np.sin(k*delta_r1)

ax.set_aspect("equal") #repère orthonormé

c= ax.imshow(W,"Blues")
fig.colorbar(c, ax=ax,label="Amplitude")
plt.show()

    Thème 4 : Mécanique

    Pendule simple non linéaire : espace des phases

    L'énergie mécanique E d'un pendule simple non linéaire de longueur \(l\) et de masse \(m\) s'écrit :
    \(E = \frac{1}{2} m l^2\dot{\theta^2} + mgl(1-cos(\theta))\)

    avec \(\dot{\theta}\) vitesse angulaire, \(\theta\) angle par rapport à la verticale et \(g\) intensité de pesanteur terrestre.

    import numpy as np
import matplotlib.pyplot as plt


theta = np.linspace(-2*np.pi,2*np.pi,100)
theta_point = np.linspace(-2*np.pi,2*np.pi,100)   
th,th_point=np.meshgrid(theta,theta_point)

E=0.5*m*l**2*th**2 + m*g*l*(1-np.cos(th_point))

fig.colorbar(c, ax=ax, label="Energie mécanique (J)")
plt.show()

    Thème 5 : Quantique

    Particule dans un réseau unidimensionnel

    Le modèle de Kronig–Penney met en jeu un réseau périodique infini de barrières de potentiel rectangulaire. Pour une période spatiale \(L\) et une barrière de largeur \(L-M\), la continuité de la fonction d'onde et de sa dérivée impose la condition suivante :
    \(cos(KL) = cos(k_{0}L)cosh(k_{1}(L-M))+\frac{k_{1}^2-k_{0}^2}{2k_{0}k_{1}}sin(k_{0}M)sinh(k_{1}(L-M))\)
    avec \(k_{0}=\sqrt{\frac{2mE}{\hbar^2}}\) et \(k_{1}=\sqrt{\frac{2m(V_{0}-E)}{\hbar^2}}\)
    où \(V_{0}\) est la hauteur de la barrière, \(E\) l'énergie et \(K\) le nombre d'onde apparaissant dans le théorème de Bloch :
    \(\psi(x+L)=e^{iKL}\psi(x)\) avec \(-\pi<KL<\pi\)

    import numpy as np
import matplotlib.pyplot as plt

L=1
M=0.4
hbar2_m=1
V0=35
E = np.linspace(0.01,125,5000,dtype=np.complex64)

k0=np.sqrt(2*E/hbar2_m)
k1=np.sqrt(2*(V0-E)/hbar2_m)

res=np.cos(k0*M)*np.cosh(k1*(L-M))+(k1**2-k0**2)/(2*k0*k1)*np.sin(k0*M)*np.sinh(k1*(L-M))
res_reel=np.real(res)
indice=np.where(np.abs(res_reel)<1)

K=np.arccos(res_reel[indice])/L

fig, ax = plt.subplots()

ax.set_xlabel(r"$K$")
ax.set_ylabel(r"$E$")
ax.set_title("Bandes d'énergie")
ax.scatter(K,E[indice],s=0.1,color="r")
ax.scatter(-K,E[indice],s=0.1,color="r")


plt.show()
